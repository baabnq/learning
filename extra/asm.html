Fibonacci

"(comment start with a quote, like in baabnq)
"memory map:
"   $00 - a
"   $01 - b
"   $02 - c

"a = 0
set 0
sRD 0

"b = 1
set 1
sRD 1

"c = 0
set 0
sRD 2

lab loop
    "c = a + b
    lDA 0
    lDR 1
    add
    sAD 2

    "a = b
    lDA 1
    sAD 0

    "b = c
    lDA 2
    sAD 1

    "print(c)
    out 2

    "if (c < 1000) goto loop
    lDA 2
    set 1000
    jmL loop
brk

Multiplication Routine

"memory map:
"   $00 - temp
"   $01 - mul.a
"   $02 - mul.b
"   $03 - mul.product

clr
set 3
add 
pha

clr
set 4
add
pha

jmS mul

pla
sAD 0
out 0
brk


"(a, b) => (a * b)
lab mul
    "deconstruct in-frame
    pla
    sAD 0
        pla
        sAD 2
        pla
        sAD 1
    lDA 0
    pha

    "multiplication loop
    clr
    sAD 3
    lab mul.loop
        "a--
        lDA 1
        jm0 mul.done
        set 1
        sub
        sAD 1

        "prod += b
        lDR 2
        lDA 3
        add
        sAD 3
        
        got mul.loop
    lab mul.done

    "reconstruct out-frame
    pla
    sAD 0
        lDA 3
        pha
    lDA 0
    pha
    ret

Recursive Factorial using Math.baabnq

How to compile and link

Assuming the factorial program is saved under fac.s1

python3 compi.py -i libs/math.baabnq -o math.s1
cat fac.s1 math.s1 > build.s1
python3 vm.py -f build.s1
        
"memory map:
"   $10 - n (the number)

"($0 - $9 are not used 
" because they are reserved by the stdlib)

clr
set 5
add
pha

jmS fac

pla
sAD 10
out 10
brk


"(n) => (!n)
lab fac
    "in-frame
    jmS Stack::Swap
        pla
        sAD 10

    "n
    "($0 will collide in recursive call)
    lDA 10
    pha


    "fac(n-1)
    lDA 10
    set 1
    jmA fac.basecase
    sub
    pha
    jmS fac

    "n * fac(n-1)
    jmS Math::Mul

    lab fac.basecase

    "out-frame
    "(out-paramter is on stack already)
    jmS Stack::Swap
    ret

Sieve of Era.. Who?

"ruffly translated from: 
"https://rosettacode.org/wiki/Sieve_of_Eratosthenes#C

"memory map:
"   $10 - n
"   (m doesn't exist because sqrt is hard)
"   $12 - tmp
"   $13 - i
"   $14 - j
"   $20 - sieve


set 10000
sRD 10


"sieve = calloc(n+1, sizeof(char));
lDR 10
ahm
sAD 20

"sieve[0] = 1;
"sieve[1] = 1;
set 1
lDA 20
sAD 12
sRP 12
add
sAD 12
sRP 12


set 2
sRD 13
lab finderLoop
    "compute position in sieve
    lDA 13
    lDR 20
    add
    sAD 12

    "check if prime by sieve
    lPA 12
    set 1
    jmA noPrime

        "if it's a prime, print it
        out 13

        "mark all multiples
        lDA 13
        sAD 14
        lab markLoop
            "compute sieve position
            lDA 14
            lDR 20
            add
            sAD 12

            "set sieve position
            set 1
            sRP 12

            "loop control
            lDA 14
            lDR 13
            add
            sAD 14
            lDR 10
        jmL markLoop

    lab noPrime

    "loop control
    lDA 13
    set 1
    add
    sAD 13
    lDR 10
jmL finderLoop

brk
